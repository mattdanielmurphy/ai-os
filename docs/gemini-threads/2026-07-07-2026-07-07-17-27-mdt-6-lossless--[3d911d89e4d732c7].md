#!/bin/bash

# Define your source and destination paths
SRC="/Users/old_user/Downloads"
DEST="/Users/new_user/Downloads"

mkdir -p "$DEST"

for file in "$SRC"/*; do
    [ -e "$file" ] || continue
    filename=$(basename "$file")
    
    # 1. Extract the raw hexadecimal Date Added attribute from the source file
    raw_added=$(xattr -p com.apple.metadata:kMDItemDateAdded "$file" 2>/dev/null)

    if [ -not -z "$raw_added" ]; then

    fi
done

To do this cleanly in macOS, we can use stat to compare the two timestamps in Unix epoch time (seconds since 1970), pick the larger (more recent) number, convert it to the format Finder expects, and inject it into the Date Added attribute.

for f in ~/Downloads/*; do
    # Skip if it's not a file/folder
    [ -e "$f" ] || continue
    
    # Get Unix timestamps: %m = modified, %B = birth/created
    mod_epoch=$(stat -f "%m" "$f")
    birth_epoch=$(stat -f "%B" "$f")
    
    # Fallback if birth time isn't supported/readable by the file system
    if [ -z "$birth_epoch" ] || [ "$birth_epoch" -eq 0 ]; then
        birth_epoch=0
    fi
    
    # Determine which timestamp is more recent
    if [ "$mod_epoch" -gt "$birth_epoch" ]; then
        target_epoch=$mod_epoch
    else
