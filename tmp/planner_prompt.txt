            function loadCustomDefaults() {
                try {
                    const saved = localStorage.getItem('university_sim_custom_defaults')
                    if (saved) return JSON.parse(saved)
                } catch (e) {
                    console.warn('Could not read custom defaults:', e)
                }
                return {}
            }

            function saveCustomDefaults(allDefaults) {
                try {
                    localStorage.setItem('university_sim_custom_defaults', JSON.stringify(allDefaults))
                } catch (e) {
                    console.warn('Could not save custom defaults:', e)
                }
            }

            function getActivePresetBaseline(presetKey) {
                const custom = loadCustomDefaults()
                return Object.assign({}, basePresets[presetKey], custom[presetKey] || {})
            }

            function formatDefVal(val, type) {
                if (type === 'currency') return '$' + Math.round(val).toLocaleString()
                if (type === 'percent') return (val % 1 === 0 ? val : val.toFixed(1)) + '%'
                return val
            }

            function checkModifiedState() {
                let modCount = 0
                const currentOtherExp = parseFloat(document.getElementById('otherExpenses').value)

                // Update benchmark chips for otherExpenses
                const chipValues = [467, 650, 850, 1100]
                chipValues.forEach((chipVal) => {
                    const chipEl = document.getElementById('chip-' + chipVal)
                    if (chipEl) {
                        if (Math.abs(currentOtherExp - chipVal) < 5) {
                            chipEl.classList.add('active')
                        } else {
                            chipEl.classList.remove('active')
                        }
                    }
                })

                trackedControls.forEach((item) => {
                    const el = document.getElementById(item.id)
                    const cg = document.getElementById('cg-' + item.id)
                    const metaRow = document.getElementById('meta-' + item.id)
                    const modBadge = document.getElementById('mod-' + item.id)

                    if (!el || !cg) return

                    const currentVal = parseFloat(el.value)
                    const defaultVal = activeDefaults[item.id] !== undefined ? activeDefaults[item.id] : currentVal
                    const isDiff = Math.abs(currentVal - defaultVal) > 0.01

                    if (isDiff) {
                        modCount++
                        cg.classList.add('is-modified')
                        if (modBadge) {
                            modBadge.innerText = 'Default: ' + formatDefVal(defaultVal, item.type)
                        }
                        if (metaRow) metaRow.classList.add('visible')
                    } else {
                        cg.classList.remove('is-modified')
                        if (metaRow) metaRow.classList.remove('visible')
                    }
                })

                const banner = document.getElementById('sandboxBanner')
                const countSpan = document.getElementById('modCount')
                if (banner && countSpan) {
                    countSpan.innerText = modCount
                    if (modCount > 0) {
                        banner.classList.add('visible')
                    } else {
                        banner.classList.remove('visible')
                    }
                }
            }

            function revertControl(id) {
                if (activeDefaults[id] !== undefined) {
                    document.getElementById(id).value = activeDefaults[id]
                    updateDashboard(true)
                }
            }

            function lockControl(id) {
                const el = document.getElementById(id)
                if (!el) return
                const newVal = parseFloat(el.value)
                activeDefaults[id] = newVal

                const allCustom = loadCustomDefaults()
                if (!allCustom[activePresetKey]) allCustom[activePresetKey] = {}
                allCustom[activePresetKey][id] = newVal
                saveCustomDefaults(allCustom)

                updateDashboard(true)
            }

            function lockAllModifiedDefaults() {
                trackedControls.forEach((item) => {
                    const el = document.getElementById(item.id)
                    if (el) {
                        activeDefaults[item.id] = parseFloat(el.value)
                    }
                })

                const allCustom = loadCustomDefaults()
                allCustom[activePresetKey] = Object.assign({}, activeDefaults)
                saveCustomDefaults(allCustom)

                updateDashboard(true)
            }

            function revertAllToDefaults() {
                loadPreset(activePresetKey)
            }

                let startSavings = parseFloat(document.getElementById('currentSavings').value) || 0

                // Effective Monthly Rent & COL
                let effectiveMonthlyRent = twoMonthsFree ? (baseRent * 10 / 12) : baseRent
                let totalMonthlyCOL = effectiveMonthlyRent + otherExpenses

                let loadSelection = document.getElementById('courseLoad').value
                let summer = document.getElementById('summerStudy').checked
                let msc = document.getElementById('mscPathway').checked
                let jobInc = parseFloat(document.getElementById('partTimeIncome').value)
                let hisaRate = parseFloat(document.getElementById('hisaRate').value) / 100.0
                let showCareer = document.getElementById('showCareer').checked
                let baseSalary = parseFloat(document.getElementById('careerSalary').value)
                let growthRate = parseFloat(document.getElementById('salaryGrowth').value) / 100.0

                // Grants Checked
                let hasCSGD = document.getElementById('grantCSGD').checked
                let hasABDis = document.getElementById('grantABDisability').checked
                let hasCSGFT = document.getElementById('grantCSGFT').checked
                let hasNewBeg = document.getElementById('grantNewBeginnings').checked
                let hasGoldStd = document.getElementById('grantGoldStandard').checked
                let hasRutherford = document.getElementById('grantRutherford').checked
                let hasNVOL = document.getElementById('grantNVOL').checked
                let hasKeyera = document.getElementById('grantKeyera').checked
                let hasGoogleLime = document.getElementById('grantGoogleLime').checked
                let hasAMIPearson = document.getElementById('grantAMIPearson').checked
                let hasCHHA = document.getElementById('grantCHHA').checked
                let hasJasonLang = document.getElementById('grantJasonLang').checked
                let hasSuppBursary = document.getElementById('grantSuppBursary').checked

                // Labels
                document.getElementById('valSavings').innerText = `$${startSavings.toLocaleString()}`
                document.getElementById('valBaseRent').innerText = `$${baseRent.toLocaleString()}`
                document.getElementById('valOtherExp').innerText = `$${Math.round(otherExpenses).toLocaleString()}`
                document.getElementById('valEffectiveRent').innerText = `$${Math.round(effectiveMonthlyRent).toLocaleString()}/mo`
                document.getElementById('valTotalCOL').innerText = `$${Math.round(totalMonthlyCOL).toLocaleString()}/mo`
                document.getElementById('valJob').innerText = `$${jobInc.toLocaleString()}`

                document.getElementById('thHisa').innerText = `@ ${(hisaRate * 100).toFixed(1)}%`

                document.getElementById('valGrowth').innerText = `${(growthRate * 100).toFixed(0)}%`

                document.getElementById('bankCardTitle').innerText = showCareer
                    ? 'Bank Cash (Year 10)'
                    : 'Cash at Graduation'
                document.getElementById('netCardTitle').innerText = showCareer
                    ? 'Net Worth (Year 10)'
                    : 'Net Worth at Graduation'

                let coursesPerSemester = loadSelection === '40' ? 2 : loadSelection === '60' ? 3 : 5
                let annualCourses = coursesPerSemester * (summer ? 3 : 2)

                let u_rem = 40 / annualCourses
                let m_rem = msc ? 2.0 : 0.0

                let bankCash = startSavings
                let year1Cash = 0
                let gradNet = 0
                let totalLoans = 0
                let gradDebtRecorded = 0
                let cashPoolAtGrad = 0
                let tableBodyHTML = ''
                let careerTime = 0.0
                let hasGraduated = false
                const EARNINGS_LIMIT = 11234

                let maxYears = showCareer ? 10 : Math.ceil(u_rem + m_rem)

                for (let yr = 1; yr <= maxYears; yr++) {
                    let u_frac = Math.min(1.0, u_rem)
                    u_rem -= u_frac

                    let m_frac = Math.min(1.0 - u_frac, m_rem)
                    m_rem -= m_frac

                    let c_frac = 1.0 - u_frac - m_frac
                    if (!showCareer && c_frac > 0 && u_frac === 0 && m_frac === 0)
                        break

                    // CALCULATIONS - Grants & Awards
                    let yrGrants = 0
                    let grantDetailLines = []

                    if (u_frac > 0) {
                        // CSG-D ($2,800/yr)
                        if (hasCSGD) {
                            let amt = 2800 * u_frac
                            yrGrants += amt
                            grantDetailLines.push('CSG Disability: $' + Math.round(amt).toLocaleString())
                        }
                        // Alberta Disability Grant ($3,000/yr)
                        if (hasABDis) {
                            let amt = 3000 * u_frac
                            yrGrants += amt
                            grantDetailLines.push('AB Disability: $' + Math.round(amt).toLocaleString())
                        }
                        // CSG-FT (Full-Time Grant: $4,200 base or $6,300 summer)
                        if (hasCSGFT) {
                            let amt = (summer ? 6300 : 4200) * u_frac
                            yrGrants += amt
                            grantDetailLines.push('CSG Full-Time: $' + Math.round(amt).toLocaleString())
                        }

                        // Year 1 Entrance & Upgrading
                        if (yr === 1) {
                            if (hasGoldStd) {
                                yrGrants += 6000
                                grantDetailLines.push('U of A Gold Standard: $6,000')
                            }
                            if (hasRutherford) {
                                yrGrants += 2500
                                grantDetailLines.push('Alexander Rutherford: $2,500')
                            }
                            if (hasNVOL) {
                                yrGrants += 1000
                                grantDetailLines.push('Adult Equivalency (NVOL Letter): $1,000')
                            }
                            if (hasNewBeg) {
                                yrGrants += 5000
                                grantDetailLines.push('New Beginnings: $5,000')
                            }
                            if (hasKeyera) {
                                yrGrants += 4500
                                grantDetailLines.push('Keyera Energy Award: $4,500')
                            }
                            if (hasGoogleLime) {
                                yrGrants += 5000
                                grantDetailLines.push('Google Lime CS: $5,000')
                            }
                            if (hasAMIPearson) {
                                yrGrants += 5000
                                grantDetailLines.push('AMI Pearson Award: $5,000')
                            }
                            if (hasCHHA) {
                                yrGrants += 2000
                                grantDetailLines.push('CHHA Disability Award: $2,000')
                            }
                        }

                        // Upper Years Continuing
                        if (yr > 1) {
                            if (hasJasonLang) {
                                let amt = 1000 * u_frac
                                yrGrants += amt
                                grantDetailLines.push('Jason Lang: $' + Math.round(amt).toLocaleString())
                            }
                            if (hasSuppBursary) {
                                let amt = 2500 * u_frac
                                yrGrants += amt
                                grantDetailLines.push('Supplementary Bursary: $' + Math.round(amt).toLocaleString())
                            }
                        }
                    }

                    // MSc Stipend
                    let stipend = 26000 * m_frac
                    if (stipend > 0) {
                        yrGrants += stipend
                        grantDetailLines.push('MSc Research Stipend: $' + Math.round(stipend).toLocaleString())
                    }

                    // LOANS ALLOCATION (with lease boost)
                    let baseLoanPerYear = summer ? 25500 : 17000
                    let leaseBoostPerYear = submitLease ? 4800 : 0 // +$400/mo shelter allowance
                    let totalLoanAvailable = (baseLoanPerYear + leaseBoostPerYear) * u_frac
                    let yrJob = jobInc * (u_frac + m_frac)

                    let limit = EARNINGS_LIMIT * u_frac
                    let loanReduction = 0
                    if (yrJob > limit) {
                        loanReduction = yrJob - limit
                        totalLoanAvailable = Math.max(0, totalLoanAvailable - loanReduction)
                    }
                    totalLoans += totalLoanAvailable

                    let yrCareer = 0
                    if (c_frac > 0 && showCareer) {
                        let currentSalary = baseSalary * Math.pow(1 + growthRate, careerTime)
                        yrCareer = currentSalary * c_frac
                        careerTime += c_frac
                    }

                    // EXPENSES CALCULATION
                    let u_tuition = annualCourses * 750 * u_frac
                    let m_tuition = 5000 * m_frac
                    let annualLiving = totalMonthlyCOL * 12 * (u_frac + m_frac + (showCareer ? c_frac : 0))
                    let tax = yrCareer * 0.3
                    let totalExpenses = u_tuition + m_tuition + annualLiving + tax

                    let totalIncome = yrGrants + totalLoanAvailable + yrJob + yrCareer
                    let yrSurplus = totalIncome - totalExpenses
                    bankCash = bankCash * (1.0 + hisaRate) + yrSurplus

                    if (yr === 1) {
                        year1Cash = bankCash
                    }

                    // FORMATTING AND TOOLTIPS
                    let title = 'Year ' + yr
                    if (c_frac > 0.99)
                        title = 'Year ' + yr + ' <span class="subtext">Career</span>'

                    else

                    let grantTT = grantDetailLines.length > 0 ? grantDetailLines.join('\n') : 'No grants active'
                    let grantHtml = '<span class="ttip">$' + Math.round(yrGrants).toLocaleString() + '<span class="ttip-text">' + grantTT + '</span></span>'

                    let loanTT = 'Base Allocation: $' + Math.round(baseLoanPerYear * u_frac).toLocaleString() + '\nLease Boost: +$' + Math.round(leaseBoostPerYear * u_frac).toLocaleString() + (loanReduction > 0 ? '\nIncome Clawback: -$' + Math.round(loanReduction).toLocaleString() : '')
